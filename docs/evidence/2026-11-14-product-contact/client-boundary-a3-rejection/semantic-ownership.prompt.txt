+// Canonical four-kind attention record key codec.
+//
+// Item identities are opaque Unicode strings that may contain slashes, percent
+// signs and astral characters, so they are transported as their complete UTF-8
+// byte sequence encoded as unpadded, canonical base64url. The codec is
+// dependency-free and safe in both Node 20 and browsers: it uses the Web
+// standard TextEncoder/TextDecoder rather than Node's Buffer.
+
+const BASE64URL_ALPHABET =
+  "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
+
+export const ATTENTION_RECORD_KINDS: readonly ["anomaly", "upcoming_cost", "contract_review", "contract_renewal"] =
+  ["anomaly", "upcoming_cost", "contract_review", "contract_renewal"];
+
+export type AttentionRecordKind = (typeof ATTENTION_RECORD_KINDS)[number];
+
+export function isAttentionRecordKind(value: unknown): value is AttentionRecordKind {
+  return (
+    value === "anomaly" ||
+    value === "upcoming_cost" ||
+    value === "contract_review" ||
+    value === "contract_renewal"
+  );

+function hasUnpairedSurrogate(value: string): boolean {
+  for (let i = 0; i < value.length; i++) {
+    const code = value.charCodeAt(i);
+    if (code >= 0xd800 && code <= 0xdbff) {
+      const next = value.charCodeAt(i + 1);
+      if (!(next >= 0xdc00 && next <= 0xdfff)) {
+        return true;
+      }
+      i++;
+    } else if (code >= 0xdc00 && code <= 0xdfff) {
+      return true;
+    }
+  }
+  return false;
+}
+
+function base64UrlEncode(bytes: Uint8Array): string {
+  let result = "";
+  for (let i = 0; i < bytes.length; i += 3) {
+    const first = bytes[i] ?? 0;
+    const second = bytes[i + 1] ?? 0;
+    const third = bytes[i + 2] ?? 0;
+    const combined = (first << 16) | (second << 8) | third;
+    result += BASE64URL_ALPHABET.charAt((combined >> 18) & 63);
+    result += BASE64URL_ALPHABET.charAt((combined >> 12) & 63);
+    if (i + 1 < bytes.length) {
+      result += BASE64URL_ALPHABET.charAt((combined >> 6) & 63);
+    }
+    if (i + 2 < bytes.length) {
+      result += BASE64URL_ALPHABET.charAt(combined & 63);
+    }
+  }
+  return result;
+}
+
+function base64UrlValue(charCode: number): number {
+  if (charCode >= 65 && charCode <= 90) {
+    return charCode - 65; // A-Z
+  }
+  if (charCode >= 97 && charCode <= 122) {
+    return charCode - 97 + 26; // a-z
+  }
+  if (charCode >= 48 && charCode <= 57) {
+    return charCode - 48 + 52; // 0-9
+  }
+  if (charCode === 45) {
+    return 62; // -
+  }
+  if (charCode === 95) {
+    return 63; // _
+  }
+  return -1;
+}
+
+function base64UrlDecode(input: string): Uint8Array | null {
+  if (input.length % 4 === 1) {
+    return null;
+  }
+  let accumulated = 0;
+  let bitCount = 0;
+  const bytes: number[] = [];
+  for (let i = 0; i < input.length; i++) {
+    const value = base64UrlValue(input.charCodeAt(i));
+    if (value < 0) {
+      return null;
+    }
+    accumulated = (accumulated << 6) | value;
+    bitCount += 6;
+    if (bitCount >= 8) {
+      bitCount -= 8;
+      bytes.push((accumulated >> bitCount) & 0xff);
+      accumulated &= (1 << bitCount) - 1;
+    }
+  }
+  if (bitCount > 0 && accumulated !== 0) {
+    return null;
+  }
+  return new Uint8Array(bytes);
+}
+
+export function encodeAttentionKey(itemId: string): string {
+  if (typeof itemId !== "string") {
+    throw new TypeError("attention key must be a string");
+  }
+  if (itemId.length === 0) {
+    throw new TypeError("attention key must not be empty");
+  }
+  if (hasUnpairedSurrogate(itemId)) {
+    throw new TypeError("attention key contains an unpaired surrogate");
+  }
+  return base64UrlEncode(new TextEncoder().encode(itemId));
+}
+
+export function decodeAttentionKey(key: unknown): string | null {
+  if (typeof key !== "string" || key.length === 0) {
+    return null;
+  }
+  const bytes = base64UrlDecode(key);
+  if (bytes === null) {
+    return null;
+  }
+  try {
+    const decoded = new TextDecoder("utf-8", {
+      fatal: true,
+      ignoreBOM: true,
+    }).decode(bytes);
+    return decoded.length === 0 ? null : decoded;
+  } catch {
+    return null;
+  }
+}

+# Four-Kind Attention Record Client Boundary
+
+**Status:** implemented (prerequisite for HLA #1593 UI; not the UI itself)
+**Surface:** `apps/web/frontend/lib/attention-key.ts`, `apps/web/frontend/lib/attention-record.ts`, `apps/web/frontend/ui-contracts/attention-current-report/**`
+
+## What this boundary provides
+
+A dependency-free, run-anywhere (Node 20 and browser) key codec plus a
+server-side record reader that fetches one attention record from its authorized
+source publication. The four kinds map one-to-one onto the generated record
+routes and their public row models:
+
+| Kind | Record route | Generated row |
+|---|---|---|
+| `anomaly` | `/reports/transaction-anomalies/records/{item_id}` | `TransactionAnomaliesRow` |
+| `upcoming_cost` | `/reports/upcoming-fixed-costs/records/{item_id}` | `UpcomingFixedCostsRow` |
+| `contract_review` | `/reports/contract-review-candidates/records/{item_id}` | `ContractReviewCandidatesRow` |
+| `contract_renewal` | `/reports/contract-renewal-watchlist/records/{item_id}` | `ContractRenewalWatchlistRow` |
+
+## Key codec
+
+`encodeAttentionKey` turns an opaque Unicode item identity into its complete
+UTF-8 bytes as canonical unpadded base64url; `decodeAttentionKey` reverses it.
+Encoding throws `TypeError` for empty, non-string, or unpaired-surrogate input
+(never silent replacement). Decoding returns `null` for any non-canonical or
+invalid input and otherwise the exact whole string — a leading U+FEFF is data,
+not a byte-order mark, and is preserved (`TextDecoder` is used with
+`ignoreBOM: true` and `fatal: true`).
+
+## Record reader
+
+`getAttentionRecord` validates kind and decodes the key **before** any request,
+so invalid values return `{ status: "invalid" }` with zero fetches. Four
+explicit literal branches keep each generated path correlated with its row
+type. Status is read before any error body: 401 redirects to `/login`, 403 is
+`denied`, 404 is `not_found`, 503 is `unavailable` (with a `retryAfter` exposed
+only for a valid delta-seconds or HTTP-date header), and every other non-OK
+status, transport failure, or malformed success envelope throws a sanitized
+error. A successful envelope must have exactly `item_id` and `rows`, the exact
+decoded identity, and only declared public fields of the correct scalar kind;
+missing or null fields stay absent/null for a future "Unknown" UI.
+
+## Scope boundaries
+
+No change to `backend.ts`, the generic auth/transport helpers, or the generated
+contract. The full four-kind queue/detail UI and HLA #1593 remain open and are
+tracked separately. The frozen contract triplet
+(`intent.md`, `baseline.tokens.json`, `ui-contract.yaml`) is copied unchanged
+from the product-contact ui-stage contracts and is existing-style reuse, not a
+claim of new human approval.

+def _key(value: str) -> str:
+    return base64.urlsafe_b64encode(value.encode("utf-8")).decode().rstrip("=")

+def _node() -> str:
+    node = shutil.which("node")
+    if not node:
+        candidates = sorted((ROOT / ".tooling").glob("node-v20*/bin/node"))
+        node = str(candidates[-1]) if candidates else None
+    assert node and Path(node).is_file(), "Node 20 unavailable on PATH or checkout tooling"
+    return node
+
+
+def _modules() -> Path:
+    modules = WEB / "node_modules"
+    if (modules / ".bin/tsc").is_file():
+        return modules
+    node_dir = Path(_node()).parent
+    npm = shutil.which("npm") or str(Path(node_dir, "npm"))
+    deps = Path(tempfile.mkdtemp(prefix="hla-worker-deps-"))
+    for name in ("package.json", "package-lock.json"):
+        shutil.copy2(WEB / name, deps / name)
+    result = subprocess.run(
+        [npm, "ci", "--no-audit", "--no-fund"],
+        cwd=deps,
+        capture_output=True,
+        text=True,
+        timeout=300,
+    )
+    assert result.returncode == 0, "locked npm ci failed: " + result.stderr[-3000:]
+    return deps / "node_modules"

+def test_frozen_style_triplet_copied_exactly() -> None:

+def test_attention_key_codec_preserves_whole_utf8_identity() -> None:
+    node = _node()
+    modules = _modules()
+    source = Path(tempfile.mkdtemp(prefix="hla-worker-codec-"))
+    output = Path(tempfile.mkdtemp(prefix="hla-worker-codec-out-"))
+    shutil.copy2(WEB / "lib/attention-key.ts", source / "attention-key.ts")
+    (source / "node_modules").symlink_to(modules, target_is_directory=True)
+    compile_result = subprocess.run(
+        [
+            str(modules / ".bin/tsc"),
+            str(source / "attention-key.ts"),
+            "--target",
+            "ES2022",
+            "--module",
+            "commonjs",
+            "--moduleResolution",
+            "node",
+            "--lib",
+            "ES2022,DOM",
+            "--types",
+            "node",
+            "--skipLibCheck",
+            "--outDir",
+            str(output),
+        ],
+        cwd=source,
+        capture_output=True,
+        text=True,
+        timeout=90,
+    )
+    assert compile_result.returncode == 0, compile_result.stdout + compile_result.stderr
+    probe = (
+        "const m=require(process.argv[1]);"
+        "const cases=['\\uFEFFa','\\uFEFF','a/b_%_雪','renewal_🙂/x','x'];"
+        "const enc=cases.map(x=>m.encodeAttentionKey(x));"
+        "const dec=enc.map(x=>m.decodeAttentionKey(x));"
+        "const invalid=['','YQ==','YR','a','bad+','_w','wK8','7aCA',null,4,[]]"
+        ".map(x=>m.decodeAttentionKey(x));"
+        "let reject=true;"
+        "for(const x of ['',42,null,{},[],'\\ud800','\\udc00'])"
+        "{try{m.encodeAttentionKey(x);reject=false}catch(e){if(!(e instanceof TypeError))throw e;}}"
+        "console.log(JSON.stringify({enc,dec,invalid,reject}));"
+    )
+    run = subprocess.run(
+        [node, "-e", probe, str(output / "attention-key.js")],
+        cwd=WEB,
+        capture_output=True,
+        text=True,
+        timeout=30,
+    )
+    assert run.returncode == 0, run.stderr
+    result = json.loads(run.stdout)
+    assert result["enc"] == [_key(case) for case in ("\uFEFFa", "\uFEFF", "a/b_%_雪", "renewal_🙂/x", "x")]
+    assert result["dec"] == ["\uFEFFa", "\uFEFF", "a/b_%_雪", "renewal_🙂/x", "x"]
+    assert result["invalid"] == [None] * 11
+    assert result["reject"] is True
