+function assertEncodableText(value: unknown): asserts value is string {
+  if (typeof value !== "string") {
+    throw new TypeError("Attention key input must be a string");
+  }
+  if (value.length === 0) {
+    throw new TypeError("Attention key input must not be empty");
+  }
+  for (let index = 0; index < value.length; index += 1) {
+    const unit = value.charCodeAt(index);
+    if (unit >= 0xd800 && unit <= 0xdbff) {
+      const next = value.charCodeAt(index + 1);
+      if (!(next >= 0xdc00 && next <= 0xdfff)) {
+        throw new TypeError("Attention key input contains an unpaired surrogate");
+      }
+      index += 1;
+    } else if (unit >= 0xdc00 && unit <= 0xdfff) {
+      throw new TypeError("Attention key input contains an unpaired surrogate");
+    }
+  }

+function encodeBytes(bytes: Uint8Array): string {
+  let output = "";
+  for (let index = 0; index < bytes.length; index += 3) {
+    const first = bytes[index] as number;
+    const second = bytes[index + 1];
+    const third = bytes[index + 2];
+    output += ALPHABET.charAt(first >> 2);
+    if (second === undefined) {
+      output += ALPHABET.charAt((first & 3) << 4);
+      break;
+    }
+    output += ALPHABET.charAt(((first & 3) << 4) | (second >> 4));
+    if (third === undefined) {
+      output += ALPHABET.charAt((second & 15) << 2);
+      break;
+    }
+    output += ALPHABET.charAt(((second & 15) << 2) | (third >> 6));
+    output += ALPHABET.charAt(third & 63);
+  }
+  return output;

+  return ALPHABET.indexOf(character);

+function decodeBytes(key: string): Uint8Array | null {
+  const remainder = key.length % 4;
+  if (remainder === 1) {
+    return null;
+  }
+  const bytes: number[] = [];
+  let index = 0;
+  while (index + 4 <= key.length) {
+    const a = sextet(key.charAt(index));
+    const b = sextet(key.charAt(index + 1));
+    const c = sextet(key.charAt(index + 2));
+    const d = sextet(key.charAt(index + 3));
+    if (a < 0 || b < 0 || c < 0 || d < 0) {
+      return null;
+    }
+    bytes.push((a << 2) | (b >> 4), ((b & 15) << 4) | (c >> 2), ((c & 3) << 6) | d);
+    index += 4;
+  }
+  if (remainder === 2) {
+    const a = sextet(key.charAt(index));
+    const b = sextet(key.charAt(index + 1));
+    if (a < 0 || b < 0 || (b & 15) !== 0) {
+      return null;
+    }
+    bytes.push((a << 2) | (b >> 4));
+  } else if (remainder === 3) {
+    const a = sextet(key.charAt(index));
+    const b = sextet(key.charAt(index + 1));
+    const c = sextet(key.charAt(index + 2));
+    if (a < 0 || b < 0 || c < 0 || (c & 3) !== 0) {
+      return null;
+    }
+    bytes.push((a << 2) | (b >> 4), ((b & 15) << 4) | (c >> 2));
+  }
+  return Uint8Array.from(bytes);

+
+**Status:** Prerequisite implemented; full UI #1593 remains open.
